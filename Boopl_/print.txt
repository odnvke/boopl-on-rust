P T; // выводит  true: # (без пробелов и переносов)

P F; // выводит false: . (без пробелов и переносов)

P S; // выводит пробел:  

P N; // переносит строку

P var_a; // выводит значение переменной: # если true, . если false

P U var_a; // выводит UTF-8 символ из ячеек(перемменных) начиная с указанной;

// возможные ошибки:
// попытка вывисти значение из не обьявленной ячейки P var_c;

// попытка вывисти значение из не обьявленной ячейки P U var_a; (после var_b ещё не обьявленны)
// UTF-8 занимает от 8 до 32 бит(переменных)
// нужно обьявить переменные специально под вывод
// symbol_0;
s_0 F; s_1 F; s_2 F; s_3 F; s_4 F; s_5 F; s_6 F; s_7 F; //...

// использование:
P U s_0;
